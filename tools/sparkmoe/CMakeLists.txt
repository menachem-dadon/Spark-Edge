set(TARGET SparkMoE)
file(STRINGS "${CMAKE_SOURCE_DIR}/UPSTREAM_BASE.txt" SPARK_EDGE_UPSTREAM LIMIT_COUNT 1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/UPSTREAM_BASE.txt")
file(STRINGS "${CMAKE_SOURCE_DIR}/VERSION" SPARKMOE_VERSION LIMIT_COUNT 1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/VERSION")

if (WIN32)
    add_executable(${TARGET} WIN32 sparkmoe.cpp)
else()
    add_executable(${TARGET} sparkmoe.cpp)
endif()
target_compile_definitions(${TARGET} PRIVATE SPARKMOE_VERSION="${SPARKMOE_VERSION}")
target_compile_definitions(${TARGET} PRIVATE SPARK_EDGE_UPSTREAM="${SPARK_EDGE_UPSTREAM}")
target_compile_features(${TARGET} PRIVATE cxx_std_17)
target_include_directories(${TARGET} PRIVATE ${CMAKE_SOURCE_DIR}/common)
target_include_directories(${TARGET} PRIVATE ${CMAKE_SOURCE_DIR}/vendor/cpp-httplib)
target_include_directories(${TARGET} PRIVATE ${CMAKE_SOURCE_DIR}/vendor)
target_link_libraries(${TARGET} PRIVATE cpp-httplib llama-common llama-ui vendor::hash)
target_sources(${TARGET} PRIVATE model-library.cpp http-facade.cpp desktop-window.cpp connections.cpp downloads.cpp engine-packages.cpp)
if(WIN32)
    option(SPARKMOE_WEBVIEW2 "Build the native SparkMoE desktop window" ON)
    set(SPARKMOE_WEBVIEW2_SDK_DIR "${CMAKE_SOURCE_DIR}/build-deps/webview2/1.0.3650.58" CACHE PATH "WebView2 SDK directory")
    if(SPARKMOE_WEBVIEW2)
        if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR CMAKE_SYSTEM_PROCESSOR MATCHES "^(ARM64|aarch64)$")
            message(FATAL_ERROR "The pinned WebView2 loader requires Windows x64. Set SPARKMOE_WEBVIEW2=OFF for browser mode.")
        endif()
        if(NOT EXISTS "${SPARKMOE_WEBVIEW2_SDK_DIR}/build/native/include/WebView2.h" OR NOT EXISTS "${SPARKMOE_WEBVIEW2_SDK_DIR}/build/native/x64/WebView2Loader.dll")
            message(FATAL_ERROR "Restore the pinned WebView2 SDK using scripts/restore-webview2.ps1, or set SPARKMOE_WEBVIEW2=OFF for browser mode")
        endif()
        target_compile_definitions(${TARGET} PRIVATE SPARKMOE_WEBVIEW2)
        target_include_directories(${TARGET} PRIVATE "${SPARKMOE_WEBVIEW2_SDK_DIR}/build/native/include")
        add_custom_command(TARGET ${TARGET} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${SPARKMOE_WEBVIEW2_SDK_DIR}/build/native/x64/WebView2Loader.dll" "$<TARGET_FILE_DIR:${TARGET}>/WebView2Loader.dll")
        install(FILES "${SPARKMOE_WEBVIEW2_SDK_DIR}/build/native/x64/WebView2Loader.dll" DESTINATION bin)
    endif()
endif()

set(SPARKMOE_BRANDING_DIR ${CMAKE_SOURCE_DIR}/assets/branding)
set(SPARKMOE_ICON_FILE ${SPARKMOE_BRANDING_DIR}/spark-edge.ico)
set(SPARKMOE_LOGO_FILE ${SPARKMOE_BRANDING_DIR}/spark-edge.png)
if (WIN32 AND EXISTS ${SPARKMOE_ICON_FILE})
    configure_file("${SPARKMOE_ICON_FILE}" SparkMoE.ico COPYONLY)
    set(SPARKMOE_ICON_RC_FILE "SparkMoE.ico")
    configure_file(sparkmoe.rc.in sparkmoe.rc @ONLY)
    target_sources(${TARGET} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/sparkmoe.rc)
endif()

if (WIN32)
    target_link_libraries(${TARGET} PRIVATE shell32 ole32 advapi32 winhttp)
endif()

if (LLAMA_TOOLS_INSTALL)
    install(TARGETS ${TARGET} RUNTIME)
endif()
